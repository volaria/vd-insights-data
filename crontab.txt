# md_stats — sunucu crontab satırları (crontab -e ile ekle)
# Yollar örnek: /opt/md_stats altına kurduysan böyle kalır, değilse uyarla.

# Her gece 06:00 (sunucu saati) — IG + YT verisi çek
0 6 * * * cd /opt/md_stats && ./venv/bin/python ig_stats.py >> data/cron.log 2>&1
5 6 * * * cd /opt/md_stats && ./venv/bin/python yt_stats.py >> data/cron.log 2>&1

# Her ayın 1'i ve 15'i — IG token yenile (60 günlük token asla dolmasın)
0 5 1,15 * * cd /opt/md_stats && ./venv/bin/python renew_ig_token.py >> data/cron.log 2>&1
